### **伪代码：球员持球决策**

代码段

```
FUNCTION DecideOnBallAction(player_role_id, observation_state):
    // 全局变量和辅助函数，用于从 observation_state 中获取信息
    // 例如: GetPlayerPosition(player_index), GetBallPosition(), GetOpponentPressure(player_index),
    //       FindOpenTeammate(options), IsInDefensiveThird(), IsInAttackingThird(), GetScore(), GetTimeLeft()

    controlled_player = observation_state.active_player // 当前控制的球员
    pressure = GetOpponentPressure(controlled_player)
    position = GetPlayerPosition(controlled_player)

    // 基于球员角色 (player_role_id) 的主要决策分支
    SWITCH player_role_id:

        // ===================================================================================
        // 角色 0: 守门员 (Goalkeeper)
        // 决策核心：安全分配球权，发起进攻
        // ===================================================================================
        CASE 0 (e_PlayerRole_GK):
            IF pressure > HIGH or NoSafeShortPassOption():
                // 高压下或无安全短传选择时，优先解围或长传打击
                target = FindBestLongPassTarget(role_CF, role_Winger) // 寻找前锋或边路空当
                RETURN action_long_pass OR action_high_pass
            ELSE:
                // 低压下，通过短传组织进攻，保持控球
                safe_teammate = FindOpenTeammate([role_CB, role_LB, role_RB, role_DM]) // 寻找空位的后卫或后腰
                RETURN action_short_pass(to=safe_teammate)
            END IF

        // ===================================================================================
        // 角色 1: 中后卫 (Centre Back)
        // 决策核心：阻止渗透，安全出球，发起组织
        // ===================================================================================
        CASE 1 (e_PlayerRole_CB):
            IF pressure > HIGH and IsInDefensiveThird():
                // 在本方危险区域受压时，首要任务是解围
                RETURN action_long_pass(direction="away_from_goal")
            ELSE IF HasSpaceToDribbleForward():
                // 在有空间时可以带球向前推进，吸引防守
                RETURN action_dribble
            ELSE:
                // 寻找安全的出球点，发动进攻
                forward_pass_option = FindOpenTeammate([role_DM, role_CM])
                IF forward_pass_option is available:
                    // 优先向前传递，打破对方第一道防线
                    RETURN action_short_pass(to=forward_pass_option)
                ELSE:
                    // 否则安全地传给边后卫或回传给门将
                    side_pass_option = FindOpenTeammate([role_LB, role_RB, role_GK])
                    RETURN action_short_pass(to=side_pass_option)
                END IF
            END IF

        // ===================================================================================
        // 角色 2 & 3: 边后卫 (Left/Right Back)
        // 决策核心：提供宽度，边路助攻，在攻防间平衡
        // ===================================================================================
        CASE 2 (e_PlayerRole_LB), CASE 3 (e_PlayerRole_RB):
            IF IsInAttackingThird():
                // 在进攻三区，主要任务是传中或配合
                IF CanCrossToStriker():
                    RETURN action_high_pass(to="penalty_area") // 高球传中
                ELSE:
                    // 寻求与边锋或中场进行配合
                    combo_option = FindOpenTeammate([role_LM, role_RM, role_AM, role_CM])
                    RETURN action_short_pass(to=combo_option)
                END IF
            ELSE IF HasSpaceToRunOnFlank():
                // 在中场有空间时，带球沿边线推进
                RETURN action_sprint
            ELSE:
                // 在后场，安全地将球传给中后卫或中场
                safe_pass_option = FindOpenTeammate([role_CB, role_DM, role_CM])
                RETURN action_short_pass(to=safe_pass_option)
            END IF

        // ===================================================================================
        // 角色 4: 防守型中场 (Defence Midfield)
        // 决策核心：保护防线，连接后场与中场，控制节奏
        // ===================================================================================
        CASE 4 (e_PlayerRole_DM):
            IF pressure > MEDIUM:
                // 受压时，优先保证球权安全，简单传球
                safe_pass_option = FindOpenTeammate([role_CB, role_LB, role_RB])
                RETURN action_short_pass(to=safe_pass_option)
            ELSE:
                // 获得空间时，转身向前，组织进攻
                forward_option = FindOpenTeammate([role_CM, role_AM, role_LM, role_RM])
                IF forward_option is available:
                    RETURN action_short_pass(to=forward_option)
                ELSE IF CanSwitchPlayToOppositeFlank():
                    // 转移球到弱侧
                    RETURN action_long_pass
                ELSE:
                    // 保持控球
                    RETURN action_short_pass(to=FindOpenTeammate([role_CM]))
                END IF
            END IF

        // ===================================================================================
        // 角色 5 & 6 & 7 & 8: 中场 (CM, LM, RM, AM)
        // 决策核心：控制节奏，串联攻防，创造机会
        // ===================================================================================
        CASE 5 (e_PlayerRole_CM), CASE 6 (e_PlayerRole_LM), CASE 7 (e_PlayerRole_RM), CASE 8 (e_PlayerRole_AM):
            // 攻击型中场(AM)更具侵略性
            is_attacking_midfielder = (player_role_id == 8)

            IF IsInAttackingThird():
                // 在进攻三区，优先创造威胁
                through_ball_option = FindTeammateRunWithoutBall([role_CF, role_LM, role_RM])
                IF through_ball_option is available:
                    // 寻找直塞球机会
                    RETURN action_short_pass(to=through_ball_option)
                ELSE IF CanShoot() AND (is_attacking_midfielder OR HasGoodShootingAngle()):
                    // AM 或有好的射门角度时，直接射门
                    RETURN action_shot
                ELSE IF CanDribbleAtDefense():
                    // 盘带突破防线
                    RETURN action_dribble
                ELSE:
                    // 传中或短传配合
                    cross_option = FindOpenTeammate([role_CF])
                    IF (player_role_id == 6 OR player_role_id == 7) AND cross_option is available: // 边中场传中
                         RETURN action_high_pass(to=cross_option)
                    ELSE:
                         RETURN action_short_pass(to=FindOpenTeammate([role_CM, role_AM]))
                    END IF
                END IF
            ELSE:
                // 在中后场，调度和推进
                forward_pass = FindOpenTeammate([role_AM, role_CF])
                IF forward_pass is available:
                    RETURN action_short_pass(to=forward_pass)
                ELSE:
                    // 控制节奏，安全传递
                    circulate_pass = FindOpenTeammate([role_DM, role_CM, role_LB, role_RB])
                    RETURN action_short_pass(to=circulate_pass)
                END IF
            END IF

        // ===================================================================================
        // 角色 9: 中锋 (Central Front)
        // 决策核心：在禁区内完成最后一击，或为队友做球
        // ===================================================================================
        CASE 9 (e_PlayerRole_CF):
            IF IsInPenaltyBox() AND HasShootingAngle():
                // 在禁区内有机会，首要选择是射门
                RETURN action_shot
            ELSE IF IsBackToGoalAndUnderPressure():
                // 背身拿球，为后上队友做球
                layoff_target = FindOpenTeammate([role_AM, role_CM], direction="behind")
                IF layoff_target is available:
                    RETURN action_short_pass(to=layoff_target)
                ELSE:
                    // 无法做球则尽力护球
                    RETURN action_dribble // 模拟护球
                END IF
            ELSE IF TeammateIsInBetterScoringPosition():
                // 如果队友位置更好，选择传球
                RETURN action_short_pass(to=teammate)
            ELSE:
                // 尝试通过盘带创造射门机会
                RETURN action_dribble
            END IF

        // ===================================================================================
        // 默认行为
        // ===================================================================================
        DEFAULT:
            // 如果没有明确指令，执行空闲动作
            RETURN action_idle

    END SWITCH

END FUNCTION
```
