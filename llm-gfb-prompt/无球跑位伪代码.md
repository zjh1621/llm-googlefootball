FUNCTION decide_GK_off_ball_move(my_player_state, game_state):
    // 现代“清道夫”守门员逻辑
    
    // 防守时：根据防线位置调整站位，准备拦截过顶球
    IF game_state.ball_owned_team == opponent_team_id:
        defensive_line_pos = get_defensive_line_height(my_team_positions)
        // 站位要比传统门将靠前，与后卫线距离缩短
        target_pos = position_behind(defensive_line_pos, distance=15m) 
        
        // 判断是否需要出击拦截过顶长传
        IF is_ball_a_through_pass_behind_defenseline() AND can_i_reach_ball_first():
            RETURN MOVE_TOWARDS(ball_pos) // 作为清道夫出击
        ELSE:
            RETURN MOVE_TO(target_pos) // 保持支援位置
    
    // 进攻时：为后卫提供安全的回传出球点
    ELSE IF game_state.ball_owned_team == my_team_id:
        ball_possessor = game_state.ball_owned_player
        ball_possessor_pos = my_team_positions[ball_possessor]
        
        // 移动到能提供安全回传角度的位置，而不是躲在防守队员身后
        target_pos = find_safe_backpass_angle_position(ball_possessor_pos)
        RETURN MOVE_TO(target_pos)
        
    // 转换或球权争夺时
    ELSE:
        // 保持一个能快速应对两种情况（进攻或防守）的居中位置
        RETURN HOLD_POSITION_AND_SCAN()
```

FUNCTION decide_CB_off_ball_move(my_player_state, game_state):
    
    // 防守时：核心是位置纪律，在压迫和保护中抉择
    IF game_state.ball_owned_team == opponent_team_id:
        partner_CB = find_teammate_with_role(e_PlayerRole_CB)
        
        // 压迫、保护、平衡原则
        IF i_am_closest_defender_to_ball() AND is_in_pressing_zone():
            RETURN PRESS_BALL_CARRIER() // 上前压迫
        ELSE IF partner_CB.is_pressing():
            RETURN PROVIDE_COVER_FOR(partner_CB) // 为上抢的搭档提供保护
        ELSE:
            // 维持防线紧凑，管理身后空间
            RETURN MAINTAIN_FORMATION_AND_MARK_NEAREST_FORWARD()
            
    // 进攻时：作为进攻发起点，为队友提供出球选择
    ELSE IF game_state.ball_owned_team == my_team_id:
        // 适度拉开，为门将或后腰创造接球角度和空间
        target_pos = find_open_space_for_build_up(my_player_state.position)
        RETURN MOVE_TO(target_pos)
        
    // 转换时
    ELSE:
        // 由攻转守：快速回追，填补中路空当
        IF just_lost_ball():
            RETURN RECOVER_TO_CENTRAL_DEFENSIVE_POSITION()
        // 由守转攻：保持位置准备发动长传反击
        ELSE:
            RETURN HOLD_POSITION_AND_SCAN_FOR_LONG_PASS()
```

FUNCTION decide_FB_off_ball_move(my_player_state, game_state):
    my_winger = find_teammate_on_my_flank_with_role(e_PlayerRole_LM/RM)

    // 防守时：扼守边路，或根据球路内收保护

        IF ball_is_on_my_flank():
            RETURN PRESS_OPPONENT_WINGER()
        ELSE:
            // 球在另一侧时，向中间收拢协助中卫

    // 进攻时：提供宽度，套边或内切
    ELSE IF game_state.ball_owned_team == my_team_id:
        // “内收型边后卫”战术
        IF team_tactic == "INVERTED_FULLBACK":
            RETURN MOVE_TO_CENTRAL_MIDFIELD_AREA()
            
        // 传统边后卫战术
        ELSE IF my_winger.is_cutting_inside():
            RETURN EXECUTE_OVERLAPPING_RUN() // 套边跑位
        ELSE IF space_is_available_inside_winger():
            RETURN EXECUTE_UNDERLAPPING_RUN() // 内切跑位
        ELSE:
            RETURN MAINTAIN_WIDTH_TO_STRETCH_DEFENSE() // 拉开宽度
    
    // 转换时
    ELSE:
        // 由攻转守：迅速回防到边路位置
        IF just_lost_ball():
            RETURN SPRINT_BACK_TO_FLANK_POSITION()
        // 由守转攻：快速沿边路前插参与反击

            RETURN SPRINT_FORWARD_ALONG_TOUCHLINE()

FUNCTION decide_DM_off_ball_move(my_player_state, game_state):
    
    // 防守时：作为后防线屏障，封堵传球线路
    IF game_state.ball_owned_team == opponent_team_id:
        // 移动到能切断对方前锋与中场联系的位置
        target_pos = position_to_block_key_passing_lanes()
        // 覆盖前插队友留下的空当
        IF fullback_or_cm_has_moved_forward():
            target_pos = cover_vacated_space(fullback_or_cm)
        RETURN MOVE_TO(target_pos)
            
    // 进攻时：串联后场与中场，控制节奏
    ELSE IF game_state.ball_owned_team == my_team_id:
        // 回撤或横向移动，为持球队友提供安全的出球点
        ball_carrier = game_state.left_team[game_state.ball_owned_player]
        target_pos = find_support_position_for(ball_carrier)
        RETURN MOVE_TO(target_pos)
        
    // 转换时
    ELSE:
        // 由攻转守：即时反抢或回撤保护关键区域
        IF just_lost_ball() AND ball_is_close():
            RETURN COUNTER_PRESS() // 即时反抢
        ELSE:
            RETURN RECOVER_TO_POSITION_IN_FRONT_OF_CBS()
        // 由守转攻：成为由守转攻的枢纽
        IF just_won_ball():
            RETURN MOVE_TO_RECEIVE_BALL_FOR_DISTRIBUTION()

FUNCTION decide_AM_off_ball_move(my_player_state, game_state):
    
    // 防守时：骚扰对方后腰或中卫出球
    IF game_state.ball_owned_team == opponent_team_id:
        opponent_pivot = find_opponent_with_role(e_PlayerRole_DM)
        RETURN APPLY_LIGHT_PRESSURE_ON(opponent_pivot)
            
    // 进攻时：在“空间”中活动，创造机会
    ELSE IF game_state.ball_owned_team == my_team_id:
        striker = find_teammate_with_role(e_PlayerRole_CF)
        
        // 寻找对方中场与后卫线之间的空间
        target_pos = find_space_between_lines(opponent_team_positions)
        // 优先在肋部区域活动
        IF is_half_space_better_option():
            target_pos = find_half_space()
            
        // 进行佯攻跑位，为前锋拉扯空间
        IF should_i_make_a_decoy_run_for(striker):
            RETURN MAKE_DECOY_RUN_AWAY_FROM_STRIKER()
        ELSE:
            RETURN MOVE_INTO(target_pos)
        
    // 转换时
    ELSE:
        // 由守转攻：快速前插，寻找反击中的致命一传或射门机会
        IF just_won_ball():
            RETURN FIND_SPACE_FOR_COUNTER_ATTACK()
        ELSE:
            RETURN START_FIRST_LINE_OF_PRESS()

FUNCTION decide_CF_off_ball_move(my_player_state, game_state):
    
    // 防守时：骚扰对方中卫，参与前场压迫
    IF game_state.ball_owned_team == opponent_team_id:
        opponent_cbs = find_opponents_with_role(e_PlayerRole_CB)
        RETURN PRESS_NEAREST_CB_TO_LIMIT_PASSING_OPTIONS()
            
    // 进攻时：核心目标是创造射门机会
    ELSE IF game_state.ball_owned_team == my_team_id:
        opponent_defensive_line_height = get_defensive_line_height(opponent_team_positions)
        
        // 跑防线身后
        IF opponent_defensive_line_height > HIGH_THRESHOLD:
            RETURN MAKE_RUN_IN_BEHIND_DEFENSE()
            
        // 拉伸防线或进行佯攻跑位
        ELSE IF should_create_space_for_am():
            // 将对方中卫带离关键位置
            RETURN RUN_TOWARDS_FLANK_TO_STRETCH_DEFENSE() 
        
        // 禁区内跑位
        ELSE IF ball_is_on_flank_and_cross_is_imminent():
            // 通过急停、变向摆脱盯防，抢占有利位置
            RETURN MAKE_UNPREDICTABLE_MOVEMENT_IN_BOX()
        ELSE:
            // 保持对中后卫的牵制
            RETURN PIN_THE_CENTER_BACKS()
    
    // 转换时
    ELSE:
        // 由守转攻：作为反击的箭头或支点
        IF just_won_ball():
            RETURN IMMEDIATELY_RUN_INTO_ATTACKING_SPACE()
        ELSE:
            RETURN LEAD_THE_PRESS_FROM_THE_FRONT()
